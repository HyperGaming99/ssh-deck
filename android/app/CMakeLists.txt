# JNI-Brücke zu nodejs-mobile (Quelltexte aus nodejs-mobile-react-native, MIT-Lizenz)
cmake_minimum_required(VERSION 3.4.1)
add_library(nodejs-mobile-react-native-native-lib SHARED src/main/cpp/native-lib.cpp src/main/cpp/rn-bridge.cpp)
include_directories(libnode/include/node/)
include_directories(src/main/cpp/)
add_library(libnode SHARED IMPORTED)
set_target_properties(libnode PROPERTIES IMPORTED_LOCATION ${CMAKE_SOURCE_DIR}/libnode/bin/${ANDROID_ABI}/libnode.so)
find_library(log-lib log)
target_link_libraries(nodejs-mobile-react-native-native-lib android libnode ${log-lib})
